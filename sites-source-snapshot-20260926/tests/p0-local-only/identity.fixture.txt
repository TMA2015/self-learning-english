import {SitesAuthentication} from '@/infrastructure/auth/sites';
import {runtimeConfig} from '@/infrastructure/runtime';
export function principal(request:{headers:Headers}){const role=request.headers.get('cookie')?.match(/batch_a_role=(admin|learner|other)/)?.[1];if(role)return {accountId:role==='other'?'user:batch-a-other':'user:batch-a-ui',roles:role==='admin'?['learner','admin']:['learner']};return new SitesAuthentication((runtimeConfig().ADMIN_EMAILS??'').split(',').map(x=>x.trim().toLowerCase()).filter(Boolean)).authenticate(request)}
export function learnerKey(request:{headers:Headers}){return principal(request)?.accountId??null}
export function isAdministrator(request:{headers:Headers}){return principal(request)?.roles.includes('admin')??false}
